Actividad 2.4. Manipular el DOM

Una página con cinco secciones y sus botones ya preparados: falta el JavaScript. En cada paso escribes las funciones de una sección para buscar elementos, cambiarlos, crear otros nuevos y borrarlos.

Lo que vas a practicar

  • Buscar elementos por ID (getElementById) y por clase (getElementsByClassName)
  • Recorrer una colección de elementos con un bucle for
  • Crear, añadir y eliminar elementos (createElement, appendChild, remove)
  • Generar colores aleatorios con Math.random()
  • Eventos del ratón: mouseenter y mouseleave

Antes de empezar

  • Teoría: 7. El árbol DOM, 6. Bucles y 8. Eventos.

  2. Sección 1: buscar por ID

    Escribe las funciones cambiarTexto() y restaurarTexto():

    • cambiarTexto() busca el párrafo textoEjemplo, cambia su texto (por ejemplo, «¡Texto modificado con éxito!») y lo pone en verde y negrita.
    • restaurarTexto() le devuelve el texto original y quita el color y la negrita (asignando una cadena vacía: style.color = '').
    Ver la solución de la sección 1 
    // =========== SECCIÓN 1: Buscar por ID ===========
function cambiarTexto() {
    // 1. Buscar elemento por su ID
    const elemento = document.getElementById('textoEjemplo');

    // 2. Verificar que existe
    if (elemento) {
        // 3. Cambiar su contenido
        elemento.textContent = '¡Texto modificado con éxito! 🎉';
        elemento.style.color = 'green';
        elemento.style.fontWeight = 'bold';
    } else {
        alert('No se encontró el elemento');
    }
}

function restaurarTexto() {
    const elemento = document.getElementById('textoEjemplo');
    if (elemento) {
        elemento.textContent = 'Este texto será modificado cuando hagas clic.';
        elemento.style.color = ''; // Quitar color
        elemento.style.fontWeight = ''; // Quitar negrita
    }
}

  3. Sección 2: buscar por clase

    Escribe cambiarColorClases() y restaurarClases(). getElementsByClassName('ejemplo') devuelve los tres párrafos en una colección: recórrela con un for y cambia a cada uno el color, el fondo, el relleno (padding) y el redondeo de las esquinas. La función de restaurar vuelve a dejarlos en azul y sin lo demás.

    Ver la solución de la sección 2 
    // =========== SECCIÓN 2: Buscar por Clase ===========
function cambiarColorClases() {
    // 1. Buscar TODOS los elementos con clase "ejemplo"
    const elementos = document.getElementsByClassName('ejemplo');

    // 2. Recorrer la colección (no es un array, es HTMLCollection)
    for (let i = 0; i < elementos.length; i++) {
        // 3. Cambiar propiedades de cada elemento
        elementos[i].style.color = 'red';
        elementos[i].style.backgroundColor = '#ffe6e6';
        elementos[i].style.padding = '10px';
        elementos[i].style.borderRadius = '5px';
    }
}

function restaurarClases() {
    const elementos = document.getElementsByClassName('ejemplo');

    for (let i = 0; i < elementos.length; i++) {
        elementos[i].style.color = 'blue';
        elementos[i].style.backgroundColor = '';
        elementos[i].style.padding = '';
        elementos[i].style.borderRadius = '';
    }
}

  4. Sección 3: crear divs con colores aleatorios

    Escribe tres funciones:

    • crearDiv(): crea un div con la clase nuevo-div, el texto «Div 1», «Div 2»… (cuenta los hijos que ya tiene el contenedor), un color de fondo aleatorio y lo añade a contenedorDivs.
    • crear5Divs(): llama cinco veces a crearDiv() con un bucle.
    • limpiarDivs(): vacía el contenedor.

    Para el color aleatorio puedes usar esta función: elige seis veces un carácter al azar entre 0 y F.

    function generarColorAleatorio() {
    // Generar un color hexadecimal aleatorio, por ejemplo "#3FA2C9"
    const letras = '0123456789ABCDEF';
    let color = '#';

    for (let i = 0; i < 6; i++) {
        color += letras[Math.floor(Math.random() * 16)];
    }

    return color;
}
    Ver la solución de la sección 3 
    // =========== SECCIÓN 3: Crear Divs con Colores Aleatorios ===========
function generarColorAleatorio() {
    // Generar un color hexadecimal aleatorio
    const letras = '0123456789ABCDEF';
    let color = '#';

    for (let i = 0; i < 6; i++) {
        color += letras[Math.floor(Math.random() * 16)];
    }

    return color;
}

function crearDiv() {
    // 1. Buscar el contenedor
    const contenedor = document.getElementById('contenedorDivs');

    // 2. Crear un nuevo elemento div
    const nuevoDiv = document.createElement('div');

    // 3. Configurar propiedades del div
    nuevoDiv.className = 'nuevo-div';
    nuevoDiv.textContent = 'Div ' + (contenedor.children.length + 1);

    // 4. Asignar color aleatorio
    const colorAleatorio = generarColorAleatorio();
    nuevoDiv.style.backgroundColor = colorAleatorio;
    nuevoDiv.style.color = '#fff'; // Texto blanco para mejor contraste

    // 5. Añadir evento de clic
    nuevoDiv.onclick = function () {
        this.style.transform = 'scale(0.95)';
        setTimeout(() => {
            this.style.transform = 'scale(1)';
        }, 200);
    };

    // 6. Insertar en el contenedor
    contenedor.appendChild(nuevoDiv);
}

function crear5Divs() {
    for (let i = 0; i < 5; i++) {
        crearDiv();
    }
}

function limpiarDivs() {
    const contenedor = document.getElementById('contenedorDivs');
    contenedor.innerHTML = ''; // Vaciar todo el contenido
}

  5. Sección 4: modificar la lista

    Escribe agregarItem() y eliminarUltimo():

    • agregarItem() lee el texto del campo nuevoItem. Si está vacío, avisa con un alert. Si no, crea un li con ese texto, lo añade a la lista, vacía el campo y vuelve a poner el cursor en él (focus()).
    • eliminarUltimo() borra el último elemento de la lista (lastElementChild). Si la lista ya está vacía, avisa.

    Si quieres, añade a cada fruta nueva un emoji al azar y haz que se borre al hacer doble clic sobre ella, como en la solución.

    Ver la solución de la sección 4 
    // =========== SECCIÓN 4: Modificar Lista ===========
function agregarItem() {
    // 1. Obtener valor del input
    const input = document.getElementById('nuevoItem');
    const texto = input.value.trim();

    // 2. Validar que no esté vacío
    if (texto === '') {
        alert('Por favor, escribe algo');
        return;
    }

    // 3. Buscar la lista
    const lista = document.getElementById('listaElementos');

    // 4. Crear nuevo elemento de lista
    const nuevoLi = document.createElement('li');
    nuevoLi.textContent = texto;

    // 5. Añadir emoji aleatorio
    const emojis = ['🍎', '🍌', '🍊', '🍇', '🍓', '🍉', '🥭', '🍍'];
    const emojiAleatorio = emojis[Math.floor(Math.random() * emojis.length)];
    nuevoLi.textContent += ' ' + emojiAleatorio;

    // 6. Añadir evento para eliminar al hacer doble clic
    nuevoLi.ondblclick = function () {
        this.remove();
    };

    // 7. Insertar en la lista
    lista.appendChild(nuevoLi);

    // 8. Limpiar el input
    input.value = '';
    input.focus(); // Poner el cursor de nuevo en el input
}

function eliminarUltimo() {
    const lista = document.getElementById('listaElementos');
    const ultimoItem = lista.lastElementChild;

    if (ultimoItem) {
        ultimoItem.remove();
    } else {
        alert('No hay elementos para eliminar');
    }
}

  6. Sección 5: cajas interactivas

    Las cajas no se eliminan: se esconden y se muestran.

    • Al pasar el ratón por encima de una caja, se esconde; al sacarlo, vuelve a aparecer. Usa los eventos mouseenter y mouseleave con addEventListener en cada caja (selecciónalas con querySelectorAll('.caja')) y cambia su style.opacity.
    • cambiarCajas(): da a cada caja un color de fondo aleatorio.
    • esconderCajas(): si las cajas se ven, las esconde, y si están escondidas, las muestra (guarda en una variable si están visibles y usa style.visibility).
    Ver la solución de la sección 5 
    // =========== SECCIÓN 5: Cajas Interactivas ===========
function cambiarCajas() {
    // Cambiar colores de las 3 cajas
    const caja1 = document.getElementById('caja1');
    const caja2 = document.getElementById('caja2');
    const caja3 = document.getElementById('caja3');

    if (caja1 && caja2 && caja3) {
        caja1.style.backgroundColor = generarColorAleatorio();
        caja2.style.backgroundColor = generarColorAleatorio();
        caja3.style.backgroundColor = generarColorAleatorio();

        // Cambiar también el color del texto para que sea visible
        caja1.style.color = 'white';
        caja2.style.color = 'white';
        caja3.style.color = 'white';
    }
}

document.querySelectorAll('.caja').forEach(caja => {
    caja.addEventListener('mouseenter', () => caja.style.opacity = '0');
    caja.addEventListener('mouseleave', () => caja.style.opacity = '1');
});

// Botón "Esconder/Mostrar": esconde o vuelve a mostrar las tres cajas
let cajasVisibles = true;

function esconderCajas() {
    cajasVisibles = !cajasVisibles; // cambia de true a false y al revés
    document.querySelectorAll('.caja').forEach(caja => {
        caja.style.visibility = cajasVisibles ? 'visible' : 'hidden';
    });
}
